rule("round change")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Global.numberofzombies <= 0;
		Global.O == False;
		Global.gamestarted == 1;
	}

	actions
	{
		Destroy All Dummy Bots;
		Wait(0.250, Ignore Condition);
		Global.roundnumber += 1;
		Global.Z = 1;
		Set Ultimate Charge(All Players(Team 1), 100);
		Big Message(All Players(All Teams), String("Round {0}", Global.roundnumber));
		Global.P = 0;
		All Dead Players(Team 1).Packapunch = 0;
		Clear Status(All Dead Players(Team 1), Burning);
		All Dead Players(Team 1).damagevariable = 100;
		Teleport(All Dead Players(Team 1), Nearest Walkable Position(Random Value In Array(All Living Players(Team 1)) + Vector(
			Random Integer(-5, 5), Empty Array, Random Integer(-5, 5))));
		Start Forcing Player To Be Hero(All Dead Players(Team 1), Hero(Ashe));
		Resurrect(All Dead Players(Team 1));
		Global.Died[0] = Vector(500, Empty Array, 500);
		Global.Died[1] = Vector(500, Empty Array, 500);
		Global.Died[2] = Vector(500, Empty Array, 500);
		Global.Died[3] = Vector(500, Empty Array, 500);
		All Players(Team 1).InfoCooldown = 0;
		All Players(Team 1).p_P = 0;
		All Players(Team 1).p_Q = 0;
		All Players(Team 1).HideHints = 0;
		Set Status(All Living Players(Team 1), Null, Phased Out, 2);
	}
}

rule("anticrash: slow motion eases in with server load (only owner of slow motion)")
{
	event
	{
		Ongoing - Global;
	}

	actions
	{
		Global.SlowFx = 100;
		Global.SlowLoad = 100;
		While(True);
			If(Global.SlowLoad < 100 ? Server Load Average > 160 : Server Load Average > 190);
				Global.SlowLoad = Max(75, Min(100, 100 - (Server Load Average - 180) * 25 / 60));
			Else;
				Global.SlowLoad = 100;
			End;
			Set Slow Motion(Min(Global.SlowFx, Global.SlowLoad));
			Wait(0.500, Ignore Condition);
		End;
	}
}

rule("Zombie chase: straight at you, pathfinding only when stuck or you are above")
{
	event
	{
		Ongoing - Each Player;
		Team 2;
		All;
	}

	conditions
	{
		Global.O == False;
		Is Alive(Event Player) == True;
	}

	actions
	{
		Event Player._extendedPlayerCollection[15] = First Of(Sorted Array(Filtered Array(All Living Players(Team 1),
			Current Array Element.p_T == 0), Distance Between(Event Player, Current Array Element)));
		If(Event Player._extendedPlayerCollection[15] == Null);
			Event Player._extendedPlayerCollection[15] = First Of(Sorted Array(All Living Players(Team 1), Distance Between(Event Player,
				Current Array Element)));
		End;
		If(Event Player._extendedPlayerCollection[15] == Null);
			Event Player._extendedPlayerCollection[0] = Null;
			Event Player._extendedPlayerCollection[16] = 0;
			Stop Throttle In Direction(Event Player);
		Else;
			Event Player._extendedPlayerCollection[20] = Distance Between(Position Of(Event Player) * Vector(1, Empty Array, 1), Position Of(
				Event Player._extendedPlayerCollection[15]) * Vector(1, Empty Array, 1));
			Event Player._extendedPlayerCollection[21] = Y Component Of(Position Of(Event Player._extendedPlayerCollection[15]))
				- Y Component Of(Position Of(Event Player));
			If(Event Player._extendedPlayerCollection[16] == 1);
				Event Player._extendedPlayerCollection[23] = Absolute Value(Event Player._extendedPlayerCollection[21]) < 1.200 && (
					Event Player._extendedPlayerCollection[20] < 3 || (
					Total Time Elapsed - Event Player._extendedPlayerCollection[17] > 3 && Is In Line of Sight(Position Of(Event Player) + Up,
					Position Of(Event Player._extendedPlayerCollection[15]) + Up, Barriers Do Not Block LOS)));
				Event Player._extendedPlayerCollection[22] = Global._extendedGlobalCollection[16] && Global._extendedGlobalCollection[10] && !Event Player._extendedPlayerCollection[23] && Total Time Elapsed - Event Player._extendedPlayerCollection[17] < 20 && Server Load < Global._extendedGlobalCollection[15] + 60;
			Else;
				If(Horizontal Speed Of(Event Player)
					< 1.500 && Event Player._extendedPlayerCollection[20] > 3 && !Event Player._extendedPlayerCollection[27] && Is On Ground(
					Event Player) && !Has Status(Event Player, Frozen) && !Has Status(Event Player, Stunned) && !Has Status(Event Player,
					Knocked Down) && !Has Status(Event Player, Asleep) && !Has Status(Event Player, Hacked));
					Event Player._extendedPlayerCollection[18] += 1;
				Else;
					Event Player._extendedPlayerCollection[18] = 0;
				End;
				If(Absolute Value(Event Player._extendedPlayerCollection[21]) > 1.800);
					Event Player._extendedPlayerCollection[19] += 1;
				Else;
					Event Player._extendedPlayerCollection[19] = 0;
				End;
				Event Player._extendedPlayerCollection[22] = Global._extendedGlobalCollection[16] && Global._extendedGlobalCollection[10] && Global._extendedGlobalCollection[14] && Total Time Elapsed - Event Player._extendedPlayerCollection[17] > 1.500 && (
					Event Player._extendedPlayerCollection[18] >= 3 || Event Player._extendedPlayerCollection[19] >= 2);
			End;
			If(Event Player._extendedPlayerCollection[22]);
				If(Event Player._extendedPlayerCollection[16] != 1);
					Event Player._extendedPlayerCollection[16] = 1;
					Event Player._extendedPlayerCollection[17] = Total Time Elapsed;
					Call Subroutine(pfLookAtDestination);
				End;
				Event Player._extendedPlayerCollection[0] = Position Of(Event Player._extendedPlayerCollection[15]);
			Else;
				If(Event Player._extendedPlayerCollection[16] != 2);
					Event Player._extendedPlayerCollection[16] = 2;
					Event Player._extendedPlayerCollection[17] = Total Time Elapsed;

	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Event Player._extendedPlayerCollection[3] == True;
	}

	actions
	{
		Event Player._extendedPlayerCollection[9] = False;
		Event Player._extendedPlayerCollection[13] = 0;
		While(Event Player._extendedPlayerCollection[3]);
			If(Event Player._extendedPlayerCollection[5] && Event Player._extendedPlayerCollection[4] != -1 && Distance Between(Position Of(
				Event Player) * Vector(1, Empty Array, 1), Event Player._extendedPlayerCollection[5] * Vector(1, Empty Array, 1))
				<= 0.600 * Max(Event Player._extendedPlayerCollection[26], 1) && Absolute Value(Y Component Of(Position Of(Event Player))
				- Y Component Of(Event Player._extendedPlayerCollection[5])) <= 1.500);
				Call Subroutine(pfProgressEventPlayer);
			Else If(Event Player._extendedPlayerCollection[5] && Distance Between((
					Event Player._extendedPlayerCollection[6] + Event Player._extendedPlayerCollection[5]) / 2, Event Player) > (Distance Between(
					Event Player._extendedPlayerCollection[6], Event Player._extendedPlayerCollection[5]) + 4) * 1.200);
				Call Subroutine(pfRecalibrate);
			End;
			If(!Event Player._extendedPlayerCollection[5] || Magnitude Of(Velocity Of(Event Player)) >= 3 * (
				Event Player._extendedPlayerCollection[14] ? Event Player._extendedPlayerCollection[14] : 1));
				Event Player._extendedPlayerCollection[13] = 0;
				Event Player._extendedPlayerCollection[9] = False;
			Else If(!Event Player._extendedPlayerCollection[13]);
				Event Player._extendedPlayerCollection[13] = Total Time Elapsed;
			Else If(Total Time Elapsed - Event Player._extendedPlayerCollection[13] > 2.100);
				Event Player._extendedPlayerCollection[9] = True;
			End;
			Wait(0.064, Ignore Condition);
		End;
	}
}

				Event Player, Stunned));

			Call Subroutine(pfRecalibrate);

		Loop If Condition Is True;
	}
